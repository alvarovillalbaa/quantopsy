cmake_minimum_required(VERSION 3.14)
project(quantopsy_trusted_cpu_scorer LANGUAGES C CXX)

if(NOT DEFINED QUANTOPSY_LLAMA_SOURCE OR NOT DEFINED QUANTOPSY_SCORER_SOURCE)
    message(FATAL_ERROR "Supply the independently verified source and authored scorer paths")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
foreach(backend CUDA MUSA HIP VULKAN KOMPUTE METAL SYCL RPC CANN OPENCL VIRTGPU)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()

add_subdirectory("${QUANTOPSY_LLAMA_SOURCE}" llama)
add_executable(quantopsy-behavioral-scorer "${QUANTOPSY_SCORER_SOURCE}")
target_link_libraries(quantopsy-behavioral-scorer PRIVATE llama)
target_compile_options(quantopsy-behavioral-scorer PRIVATE -Wall -Wextra -Werror -fno-fast-math)
